File-system operations, synchronous and asynchronous file handling and the command-line Notes App.

read-sync.js
const fs = require('fs');

const data = fs.readFileSync('sample.txt', 'utf8');

console.log(data);

console.log('This line runs AFTER the file has been fully read.');
read-async.js
const fs = require('fs');

fs.readFile('sample.txt', 'utf8', (err, data) => {
    if (err) {
        console.error('Error reading file:', err.message);
        return;
    }

    console.log(data);
});

console.log('This line runs BEFORE the file content is printed.');
write-file.js
const fs = require('fs');

fs.writeFile('output.txt', 'This is my second message!', (err) => {
    if (err) throw err;
    console.log('File written successfully.');
});

});

add-note.js
const fs = require('fs');

const note = process.argv[2];

if (!note) {
    console.log('Please provide a note. Example: node add-note.js "Buy groceries"');
    process.exit(1);
}

const entry = `[${new Date().toLocaleString()}] ${note}\n`;

fs.appendFile('notes.txt', entry, (err) => {
    if (err) throw err;
    console.log('Note added!');
});
